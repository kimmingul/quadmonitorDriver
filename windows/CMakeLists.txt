cmake_minimum_required(VERSION 3.21)
project(QuadMonitorCore LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(codec "${CMAKE_CURRENT_LIST_DIR}/../App/Sources/CFrameEncoder")
add_library(quad_frame_core STATIC "${codec}/FrameEncoder.c" "${codec}/FrameValidator.c")
target_include_directories(quad_frame_core PUBLIC "${codec}/include")
if(WIN32)
  target_compile_definitions(quad_frame_core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(quad_frame_core PUBLIC Threads::Threads)
  if(NOT APPLE)
    target_link_libraries(quad_frame_core PUBLIC m)
  endif()
endif()
if(MSVC)
  target_compile_options(quad_frame_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(quad_frame_core PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
endif()
enable_testing()
add_executable(quad_core_tests tests/core_tests.cpp)
target_link_libraries(quad_core_tests PRIVATE quad_frame_core)
foreach(test concurrent_init vendor_frames encoder guards)
  add_test(NAME ${test} COMMAND quad_core_tests ${test} "${CMAKE_CURRENT_LIST_DIR}/tests/fixtures")
endforeach()
